<div class="section-header">

<h2 class="section-title">Cerita Konsep</h2>
</div>

<div class="card">
<h3>Struk yang Hilang di Kedai Kopi Nusantara</h3>
<p>Pak Budi memesan <strong>2 cangkir kopi</strong> di Kedai Kopi Nusantara. Setiap cangkir berharga <strong>Rp 25.000</strong>, dan kedai menambahkan <strong>biaya layanan Rp 5.000</strong> sekali transaksi. Total yang harus dibayar Pak Budi adalah…</p>
<div class="story-reveal">
<p>…<strong class="num">Rp 55.000</strong>. Struk dicetak, lalu jatuh ke lantai dan hilang. Kini kasir hanya melihat angka <strong class="num">Rp 55.000</strong> tercetak di mesin. <em>Berapa cangkir yang sebenarnya dipesan Pak Budi?</em></p>
</div>

<div class="misconception">
<i class="fa-solid fa-triangle-exclamation"></i>
<div>
<strong>Miskonsepsi umum:</strong> Banyak siswa mengira "invers" berarti <code>1/f(x)</code> atau "dibalik penyebutnya". Padahal, invers adalah membalik <strong>proses</strong>: jika <code>f</code> mengubah 2 cangkir menjadi Rp 55.000, maka <code>f⁻¹</code> mengubah Rp 55.000 kembali menjadi 2 cangkir.
</div>
</div>

<div class="triggers">
<h3>Pertanyaan Pemantik</h3>
<ol>
<li>Jika kamu hanya tahu total bayar, bagaimana cara menemukan jumlah cangkir yang dipesan?</li>
<li>Apakah setiap fungsi bisa dibalik? Kapan sebuah fungsi tidak punya invers?</li>
<li>Apakah <code>f(f⁻¹(x))</code> sama dengan <code>x</code>? Mengapa demikian?</li>
</ol>
</div>

<div class="struk-machine">
<h3>Mesin Struk Otomatis</h3>
<p>Tekan tombol untuk menjalankan mesin dan menyimak tiap baris struk mucul tahap demi tahap.</p>

</div>

<div class="conclusion-panel" id="conclusion" style="display:none">
<h4><i class="fa-solid fa-lightbulb"></i> Notasi & Aturan Formal</h4>
<div class="formula-block">
<p data-step="1"><strong>Langkah 1:</strong> f(x) = 25.000x + 5.000</p>
<p data-step="2"><strong>Langkah 2:</strong> f⁻¹(y) = (y − 5.000) / 25.000</p>
<p data-step="3"><strong>Langkah 3:</strong> f⁻¹(55.000) = (55.000 − 5.000) / 25.000 = 2 cangkir</p>
</div>
</div>
</div>

<div class="summary-panel">
<h3><i class="fa-solid fa-clipboard-list"></i> Tiga Poin Inti</h3>
<ol>
<li>Invers membalik proses: <code>input → f → output</code> menjadi <code>output → f⁻¹ → input</code>.</li>
<li>Notasi <code>f⁻¹</code> dibaca "f invers", <strong>bukan</strong> "1/f" atau "f pangkat minus satu".</li>
<li>Syarat: f harus satu-satu (one-to-one); setiap output hanya berasal dari satu input.</li>
</ol>
</div>

function animateConfetti() {
cctx.clearRect(0, 0, confettiCanvas.width, confettiCanvas.height);
particles = particles.filter(p => {
p.x += p.vx;
p.y += p.vy;
p.vy += p.g;
p.angle += p.spin;
p.life -= 0.012;
if (p.life <= 0 || p.y > confettiCanvas.height + 30) return false;
cctx.save();
cctx.translate(p.x, p.y);
cctx.rotate(p.angle);
cctx.globalAlpha = Math.max(0, p.life);
cctx.fillStyle = p.color;
if (p.shape === 'rect') {
cctx.fillRect(-p.size/2, -p.size/2, p.size, p.size * 1.5);
} else {
cctx.beginPath();
cctx.arc(0, 0, p.size/2, 0, Math.PI * 2);
cctx.fill();
}
cctx.restore();
return true;
});
if (particles.length > 0) {
requestAnimationFrame(animateConfetti);
} else {
rafRunning = false;
}

function countUp(el, target, duration = 800, formatter = (x) => Math.round(x).toString()) {
if (STATE.reduceMotion) { el.textContent = formatter(target); return; }
const start = parseFloat(el.dataset.current || 0);
const startTime = performance.now();
function tick(now) {
const t = Math.min((now - startTime) / duration, 1);
const eased = 1 - Math.pow(1 - t, 3);
const val = start + (target - start) * eased;
el.textContent = formatter(val);
el.dataset.current = val;
if (t < 1) requestAnimationFrame(tick);
else el.dataset.current = target;
}
requestAnimationFrame(tick);
}

const Calc = {
tokenize(expr) {
const tokens = [];
let i = 0;
while (i < expr.length) {
const c = expr[i];
if (/\s/.test(c)) { i++; continue; }
if (/[0-9.]/.test(c)) {
let num = '';
while (i < expr.length && /[0-9.]/.test(expr[i])) { num += expr[i]; i++; }
tokens.push({ type: 'num', value: parseFloat(num) });
} else if (/[a-zA-Z]/.test(c)) {
let word = '';
while (i < expr.length && /[a-zA-Z]/.test(expr[i])) { word += expr[i]; i++; }
if (['sqrt','sin','cos','tan','log','ln','abs'].includes(word)) {
tokens.push({ type: 'func', value: word });
} else {
tokens.push({ type: 'var', value: word });
}
} else if ('+-*/^()'.includes(c)) {
const prev = tokens[tokens.length - 1];
if (c === '-' && (!prev || prev.type === 'op' || (prev.type === 'op' && prev.value !== ')'))) {
tokens.push({ type: 'num', value: 0 });
}
tokens.push({ type: 'op', value: c });
i++;
} else { i++; }
}
return tokens;
},
toRPN(tokens) {
const out = [];
const stack = [];
const prec = { '+': 1, '-': 1, '*': 2, '/': 2, '^': 3 };
const rightAssoc = { '^': true };
for (const t of tokens) {
if (t.type === 'num' || t.type === 'var') out.push(t);
else if (t.type === 'func') stack.push(t);
else if (t.type === 'op') {
while (stack.length) {
const top = stack[stack.length - 1];
if (top.type === 'func' || (top.type === 'op' && (prec[top.value] > prec[t.value] || (prec[top.value] === prec[t.value] && !rightAssoc[t.value])))) {
out.push(stack.pop());
} else break;
}
stack.push(t);
} else if (t.value === '(') stack.push(t);
else if (t.value === ')') {
while (stack.length && stack[stack.length - 1].value !== '(') out.push(stack.pop());
if (stack.length) stack.pop();
if (stack.length && stack[stack.length - 1].type === 'func') out.push(stack.pop());
}
}
while (stack.length) out.push(stack.pop());
return out;
},
evalRPN(rpn, x) {
const st = [];
for (const t of rpn) {
if (t.type === 'num') st.push(t.value);
else if (t.type === 'var') {
if (t.value === 'x') st.push(x);
else if (t.value === 'pi') st.push(Math.PI);
else if (t.value === 'e') st.push(Math.E);
else throw new Error('Variabel tak dikenal: ' + t.value);
} else if (t.type === 'func') {
if (st.length < 1) throw new Error('Argumen kurang untuk ' + t.value);
const a = st.pop();
let r;
switch (t.value) {
case 'sqrt': if (a < 0) throw new Error('Akar negatif tak terdefinisi'); r = Math.sqrt(a); break;
case 'sin': r = Math.sin(a); break;
case 'cos': r = Math.cos(a); break;
case 'tan': r = Math.tan(a); break;
case 'log': if (a <= 0) throw new Error('Logaritma harus positif'); r = Math.log10(a); break;
case 'ln': if (a <= 0) throw new Error('Ln harus positif'); r = Math.log(a); break;
case 'abs': r = Math.abs(a); break;
default: throw new Error('Fungsi tak dikenal');
}
if (isNaN(r) || !isFinite(r)) throw new Error('Hasil tak terdefinisi');
st.push(r);
} else if (t.type === 'op') {
if (st.length < 2) throw new Error('Operand kurang');
const b = st.pop(), a = st.pop();
let r;
switch (t.value) {
case '+': r = a + b; break;
case '-': r = a - b; break;
case '*': r = a * b; break;
case '/': if (b === 0) throw new Error('Pembagian nol'); r = a / b; break;
case '^': r = Math.pow(a, b); break;
default: throw new Error('Operator tak dikenal');
}
if (isNaN(r) || !isFinite(r)) throw new Error('Hasil tak terdefinisi');
st.push(r);
}
}
if (st.length !== 1) throw new Error('Sintaks tidak lengkap');
return st[0];
},
evaluate(expr, x) {
const tokens = this.tokenize(expr);
if (tokens.length === 0) throw new Error('Rumus kosong');
const rpn = this.toRPN(tokens);
return this.evalRPN(rpn, x);
},

s = s.replace(/x/g, '<span class="var">x</span>');
return `${name}(${s.replace(/<span class="var">x<\/span>/, '<span class="var">x</span>')}) = ${s}`;
},

const Quiz = {
shuffled: [],
currentIdx: 0,
score: 0,
streak: 0,
answered: false,
init() {
this.renderStart();
},
renderStart() {
const area = document.getElementById('quiz-area');
area.innerHTML = `
<div class="quiz-start-card">
<h3><i class="fa-solid fa-clipboard-question"></i> Sebelum Mulai Kuis</h3>
<p>Isi nama dan pilih kelas terlebih dahulu. Kuis terdiri dari 10 soal pilihan ganda.</p>
<div class="rules">
<ul>
<li><strong>10 soal</strong> pilihan ganda, urutan soal dan opsi diacak setiap sesi.</li>
<li>Satu kesempatan per soal — begitu pilihan diklik, semua terkunci.</li>
<li>Skor dan rentetan benar beruntun tampil otomatis.</li>
<li>Pembahasan penuh tersedia setelah kuis selesai.</li>
</ul>
</div>
<label for="qz-name">Nama Lengkap</label>
<input type="text" id="qz-name" placeholder="Min. 2 huruf" autocomplete="name">
<label for="qz-class">Kelas</label>
<select id="qz-class">
<option value="">— Pilih kelas —</option>
${CONFIG.kelasList.map(k => `<option>${k}</option>`).join('')}
</select>
<button class="btn-primary" id="qz-start-btn" style="margin-top:1rem"><i class="fa-solid fa-play"></i> Mulai Kuis</button>
</div>
`;
document.getElementById('qz-start-btn').addEventListener('click', () => this.start());
document.getElementById('qz-name').focus();
},
start() {
const name = document.getElementById('qz-name').value.trim();
const kelas = document.getElementById('qz-class').value;
const nameInput = document.getElementById('qz-name');
const classInput = document.getElementById('qz-class');
if (name.length < 2) {
nameInput.classList.add('shake');
showToast('Isi nama minimal 2 huruf', 'fa-triangle-exclamation');
playWrong();
setTimeout(() => nameInput.classList.remove('shake'), 500);
return;
}
if (!kelas) {
classInput.classList.add('shake');
showToast('Pilih kelas terlebih dahulu', 'fa-triangle-exclamation');
playWrong();
setTimeout(() => classInput.classList.remove('shake'), 500);
return;
}
STATE.quizInProgress = true;
STATE.quizScore = 0;
STATE.quizAnswers = [];
Calc.lock();
// Shuffle questions and options
this.shuffled = QUIZ_DATA.map(q => {
const correctText = q.options[q.key];
const shuffledOpts = [...q.options].sort(() => Math.random() - 0.5);
return { ...q, options: shuffledOpts, key: shuffledOpts.indexOf(correctText) };
}).sort(() => Math.random() - 0.5);
this.currentIdx = 0;
this.score = 0;
this.streak = 0;
this.renderQuestion();
playClick();
},
renderQuestion() {
const q = this.shuffled[this.currentIdx];
const area = document.getElementById('quiz-area');
this.answered = false;
area.innerHTML = `
<div class="quiz-card">
<div class="quiz-progress">
<span class="qnum">Soal ${this.currentIdx + 1} dari 10</span>
<span class="qnum">Skor: ${this.score}/10</span>
<span class="streak" id="streak-badge" style="${this.streak >= 2 ? 'display:inline-flex' : ''}">
<i class="fa-solid fa-fire"></i> ${this.streak} beruntun
</span>
</div>
<div class="quiz-question">${q.q}</div>
<div class="quiz-options" id="quiz-options">
${q.options.map((opt, i) => `
<button class="quiz-option" data-idx="${i}">
<span class="opt-letter">${String.fromCharCode(65 + i)}</span>
<span>${opt}</span>
</button>
`).join('')}
</div>
<div id="quiz-feedback-area"></div>
</div>
`;
area.querySelectorAll('.quiz-option').forEach(btn => {
btn.addEventListener('click', () => this.answer(parseInt(btn.dataset.idx)));
});
// Scroll ke atas kuis
document.getElementById('kuis').scrollIntoView({ behavior: 'smooth', block: 'start' });
},
answer(idx) {
if (this.answered) return;
this.answered = true;
const q = this.shuffled[this.currentIdx];
const correct = idx === q.key;
const options = document.querySelectorAll('.quiz-option');
options.forEach(o => o.classList.add('locked'));
const selected = options[idx];
STATE.quizAnswers.push({
question: q.q,
options: q.options,
key: q.key,
studentIdx: idx,
correct,
explain: q.explain
});
const fbArea = document.getElementById('quiz-feedback-area');
if (correct) {
selected.classList.add('correct');
this.score++;
STATE.quizScore = this.score;
this.streak++;
playCorrect();
// Confetti from the button
const rect = selected.getBoundingClientRect();
confettiExplode(rect.left + rect.width/2, rect.top + rect.height/2, 25);
// Score animation
const scoreEl = document.querySelector('.quiz-progress .qnum:nth-child(2)');
scoreEl.style.transition = 'transform 0.3s';
scoreEl.style.transform = 'scale(1.3)';
setTimeout(() => scoreEl.style.transform = 'scale(1)', 300);
fbArea.innerHTML = `
<div class="quiz-badge correct"><i class="fa-solid fa-check"></i> Tepat!</div>
<div class="quiz-explanation">
<h4><i class="fa-solid fa-lightbulb"></i> Penjelasan Konsep</h4>
<p>${q.explain}</p>
</div>
`;
} else {
selected.classList.add('wrong');
this.streak = 0;
playWrong();
fbArea.innerHTML = `
<div class="quiz-badge wrong"><i class="fa-solid fa-xmark"></i> Belum tepat</div>
<div class="quiz-explanation">
<h4><i class="fa-solid fa-lightbulb"></i> Penjelasan Konsep</h4>
<p>${q.explain}</p>
</div>
`;
}
// Show streak badge if >= 2
const streakBadge = document.getElementById('streak-badge');
if (this.streak >= 2) {
streakBadge.classList.add('show');
streakBadge.innerHTML = `<i class="fa-solid fa-fire"></i> ${this.streak} beruntun`;
} else {
streakBadge.classList.remove('show');
}
// Add next button
const isLast = this.currentIdx === this.shuffled.length - 1;
fbArea.innerHTML += `
<button class="quiz-next-btn" id="next-btn" autofocus>
${isLast ? '<i class="fa-solid fa-flag-checkered"></i> Lihat Hasil' : '<i class="fa-solid fa-arrow-right"></i> Lanjut ke Soal berikutnya'}
</button>
`;
setTimeout(() => {
const nb = document.getElementById('next-btn');
nb.focus();
nb.addEventListener('click', () => {
if (isLast) this.showResults();
else { this.currentIdx++; this.renderQuestion(); }
});
}, 100);
playClick();
},
showResults() {
STATE.quizInProgress = false;
Calc.unlock();
const pct = Math.round((this.score / 10) * 100);
const area = document.getElementById('quiz-area');
let icon, label, color;
if (pct >= 80) { icon = 'fa-trophy'; label = 'Sangat Baik'; color = 'var(--gold)'; }
else if (pct >= 60) { icon = 'fa-medal'; label = 'Baik'; color = 'var(--correct)'; }
else if (pct >= 40) { icon = 'fa-circle-check'; label = 'Cukup'; color = 'var(--orange)'; }
else { icon = 'fa-rotate-right'; label = 'Ayo Ulangi'; color = 'var(--wrong)'; }
// Confetti rain if >= 50%
if (pct >= 50) {
confettiRain(60);
playSuccess();
}
// Unlock reflection fields
document.getElementById('refl-name').disabled = false;
document.getElementById('refl-class').disabled = false;
// Auto-fill name from quiz
const qname = document.getElementById('qz-name').value.trim();
if (qname) document.getElementById('refl-name').value = qname;
document.getElementById('refl-score').value = `${this.score} dari 10 benar (${pct}%)`;
const chipsHtml = STATE.quizAnswers.map((a, i) =>
`<div class="quiz-chip ${a.correct ? 'correct' : 'wrong'}">${i+1}</div>`
).join('');
area.innerHTML = `
<div class="quiz-result-card">
<div class="quiz-result-icon" style="color:${color}"><i class="fa-solid ${icon}"></i></div>
<div class="quiz-result-score">
<div class="big">${label}</div>
<div><span class="num">${this.score} dari 10</span> benar</div>
<div class="pct">Ketuntasan: ${pct}%</div>
</div>
<div class="quiz-chips">${chipsHtml}</div>
<div class="quiz-result-buttons">
<button class="btn-primary" id="review-btn"><i class="fa-solid fa-list"></i> Pembahasan</button>
<button class="btn-secondary" id="repeat-btn"><i class="fa-solid fa-rotate-right"></i> Ulangi</button>
<a href="#refleksi" class="btn-secondary" style="text-decoration:none"><i class="fa-solid fa-feather"></i> Refleksi</a>
</div>
</div>
<div class="quiz-review" id="quiz-review">
<h3 style="margin-bottom:1rem;color:var(--coffee)"><i class="fa-solid fa-list-ol"></i> Pembahasan Tiap Soal</h3>
${STATE.quizAnswers.map((a, i) => `
<div class="quiz-review-item ${a.correct ? 'correct-item' : 'wrong-item'}">
<h4>Soal ${i+1}</h4>
<div class="review-q">${a.question}</div>
<div class="review-ans">
<span class="your ${a.correct ? 'correct' : ''}">Jawabanmu: ${a.studentIdx >= 0 ? a.options[a.studentIdx] : '—'}</span>
<span class="key">Kunci: ${a.options[a.key]}</span>
</div>
<div class="review-explain">${a.explain}</div>
</div>
`).join('')}
</div>
`;
document.getElementById('review-btn').addEventListener('click', () => {
const r = document.getElementById('quiz-review');
r.classList.toggle('show');
if (r.classList.contains('show')) {
r.scrollIntoView({ behavior: 'smooth', block: 'start' });
playClick();
}
});
document.getElementById('repeat-btn').addEventListener('click', () => {
this.renderStart();
playClick();
});
}
};
